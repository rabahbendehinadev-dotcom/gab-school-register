function sendTelegramMessage(string $text): void {
    if (!TELEGRAM_BOT_TOKEN || !TELEGRAM_CHAT_ID) return;
    $url = 'https://api.telegram.org/bot' . TELEGRAM_BOT_TOKEN . '/sendMessage';
    $data = ['chat_id' => TELEGRAM_CHAT_ID, 'text' => $text];
    $options = ['http' => ['header' => "Content-type: application/x-www-form-urlencoded\r\n", 'method' => 'POST', 'content' => http_build_query($data), 'timeout' => 5]];
    $result = @file_get_contents($url, false, stream_context_create($options));
    file_put_contents(__DIR__ . '/telegram_log.txt', date('Y-m-d H:i:s') . ' | ' . ($result ?: 'ERROR') . "\n", FILE_APPEND);
}

function sendMakeWebhook(array $data): void {
    if (!MAKE_WEBHOOK_URL) return;
    $ch = curl_init(MAKE_WEBHOOK_URL);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
    curl_setopt($ch, CURLOPT_TIMEOUT, 5);
    curl_exec($ch);
    curl_close($ch);
}

<script>
    // تحميل الشعار الحقيقي من السيرفر
    (function () {
        var img = document.getElementById('logoImg');
        var svg = document.getElementById('logoSvg');
        if (!img || !svg) return;
        img.onload  = function () { svg.style.display = 'none'; img.style.display = 'block'; };
        img.onerror = function () { img.style.display = 'none'; svg.style.display = 'block'; };
    })();

    // ═══════════════════════════════════════════════════════
    // Gallery fade-in — آمن 100%:
    // الصور تظهر بدون CSS opacity:0
    // JavaScript يُضيف will-animate أولاً ثم يُشغّل المراقب
    // إذا فشل أي شيء → safety timeout يُظهر كل الصور بعد 1.5 ثانية
    // ═══════════════════════════════════════════════════════
    (function () {
        var cards = document.querySelectorAll('.photo-card');
        if (!cards.length) return;

        // أضف الكلاس عبر JS فقط — الصور ظاهرة بدون JS
        cards.forEach(function (c) { c.classList.add('will-animate'); });

        function revealCard(el, delay) {
            setTimeout(function () { el.classList.add('visible'); }, delay || 0);
        }

        // إذا المتصفح لا يدعم IntersectionObserver → أظهر الكل فوراً
        if (!window.IntersectionObserver) {
            cards.forEach(function (c) { revealCard(c, 0); });
            return;
        }

        var allCards = Array.from(cards);
        var observer = new IntersectionObserver(function (entries) {
            entries.forEach(function (entry) {
                if (entry.isIntersecting) {
                    var idx   = allCards.indexOf(entry.target);
                    var delay = (idx % 8) * 70;
                    revealCard(entry.target, delay);
                    observer.unobserve(entry.target);
                }
            });
        }, { threshold: 0.04, rootMargin: '0px 0px 80px 0px' });

        cards.forEach(function (c) { observer.observe(c); });

        // شبكة أمان: بعد 1.5 ثانية أظهر أي بطاقة لا تزال مخفية
        setTimeout(function () {
            document.querySelectorAll('.photo-card.will-animate:not(.visible)').forEach(function (c) {
                c.classList.add('visible');
            });
        }, 1500);
    })();
</script>
